rules_version = '2';

// ============================================================
//  Reglas de seguridad de Firestore · downloaderV
//  Pega esto en: Firebase Console > Firestore Database > Reglas
//  y pulsa "Publicar".
// ============================================================

service cloud.firestore {
  match /databases/{database}/documents {

    // (Opcional) Para que tu panel de administración pueda leer/responder
    // TODOS los chats, pon aquí tu UID de administrador. Si por ahora lo
    // gestionas desde la consola de Firebase, déjalo tal cual.
    function esAdmin() {
      return request.auth != null && request.auth.uid == 'PON_AQUI_TU_UID_ADMIN';
    }

    // --- Registro de visitas ---
    // Cualquiera puede CREAR su visita y ACTUALIZARLA (para guardar la
    // duración y si pulsó descargar). Nadie puede LEERLAS ni BORRARLAS desde
    // el cliente: eso solo tú, desde la consola o con el Admin SDK.
    match /visitas/{doc} {
      allow create: if true;
      allow update: if true;
      allow read, delete: if false;
    }

    // --- Soporte: cada usuario solo gestiona SU propio chat ---
    // La sesión se llama igual que el UID anónimo del visitante, así que
    // nadie puede leer ni borrar el chat de otra persona.
    match /soporte/{uid} {
      allow read, create, update, delete:
        if (request.auth != null && request.auth.uid == uid) || esAdmin();

      match /mensajes/{msgId} {
        allow read, create, delete:
          if (request.auth != null && request.auth.uid == uid) || esAdmin();
        allow update: if esAdmin();
      }
    }

  }
}
